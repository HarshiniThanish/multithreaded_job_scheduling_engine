cmake_minimum_required(VERSION 3.20)
project(job_scheduler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
add_executable(job_scheduler src/main.cpp)
target_include_directories(job_scheduler PRIVATE include)
target_link_libraries(job_scheduler PRIVATE Threads::Threads SQLite::SQLite3)
